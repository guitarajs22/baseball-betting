{# Wind diagram macro — renders a top-down baseball field SVG with a wind arrow. Usage: {% from "wind_diagram.html" import wind_diagram %} {{ wind_diagram(game, size=160) }} Parameters: game — a Game ORM object with .wind_deg, .wind_speed_mph, .wind_direction, .ballpark size — pixel size of the square SVG (default 160) #} {% macro wind_diagram(game, size=160) %} {% set park = game.ballpark %} {% set has_wind = game.wind_deg is not none and game.wind_speed_mph is not none %} {% set is_dome = park and park.roof_type == 'dome' %} {% set cf_deg = park.cf_bearing_deg if (park and park.cf_bearing_deg is not none) else 300 %} {# field_rel_deg: direction wind is BLOWING toward, relative to the field diagram where 0° = toward CF (out), 90° = L→R crosswind, 180° = toward HP (in), 270° = R→L crosswind. wind_deg is the compass direction wind comes FROM. We add 180 to get the TO direction, then subtract the CF bearing to make it field-relative. #} {% if has_wind and not is_dome %} {% set wind_to = (game.wind_deg + 180) % 360 %} {% set field_rel = (wind_to - cf_deg + 360) % 360 %} {# Pick arrow color: toward CF = green (helps HRs), toward HP = red (hurts), crosswind = yellow #} {% if field_rel < 45 or field_rel >= 315 %} {% set arrow_color = "#4ade80" %} {# green — out to CF #} {% set wind_label = "Out to CF" %} {% elif field_rel >= 135 and field_rel < 225 %} {% set arrow_color = "#f87171" %} {# red — in from CF #} {% set wind_label = "In from CF" %} {% elif field_rel >= 45 and field_rel < 135 %} {% set arrow_color = "#facc15" %} {# yellow — L→R crosswind #} {% set wind_label = "L→R Wind" %} {% else %} {% set arrow_color = "#facc15" %} {# yellow — R→L crosswind #} {% set wind_label = "R→L Wind" %} {% endif %} {% else %} {% set field_rel = 0 %} {% set arrow_color = "#6b7280" %} {% set wind_label = "No data" %} {% endif %}
{% if has_wind and not is_dome %}
{{ wind_label }} {{ "%.1f"|format(game.wind_speed_mph) }} mph {{ game.wind_direction or '' }}
{% endif %}
{% endmacro %}
